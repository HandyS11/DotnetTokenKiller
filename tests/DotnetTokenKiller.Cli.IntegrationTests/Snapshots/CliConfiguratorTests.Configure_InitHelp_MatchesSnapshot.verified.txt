DESCRIPTION:
Install dtk integration artifacts for an AI assistant provider

USAGE:
    dtk init <provider> [OPTIONS]

EXAMPLES:
    dtk init claude
    dtk init claude --dir /path/to/project --force
    dtk init copilot
    dtk init copilot-cli
    dtk init copilot-cli --global
    dtk init gemini
    dtk init codex
    dtk init codex --global
    dtk init opencode
    dtk init opencode --global
    dtk init antigravity
    dtk init antigravity --global
    dtk init cursor
    dtk init windsurf
    dtk init aider
    dtk init jetbrains
    dtk init claude --uninstall
    dtk init codex --global --uninstall

ARGUMENTS:
    <provider>    AI assistant provider to set up (claude, copilot, copilot-cli, gemini, codex,     
                  opencode, antigravity, cursor, windsurf, aider, jetbrains)                        

OPTIONS:
    -h, --help         Prints help information                                                      
    -d, --dir          Target project root directory (default: current directory)                   
    -f, --force        Overwrite existing files instead of skipping them                            
    -g, --global       Install into the user's home config (claude, copilot-cli, gemini, codex,     
                       opencode, antigravity, aider) instead of a project                           
        --uninstall    Remove what 'dtk init' installed for the provider instead (respects --dir and
                       --global); edited files are kept
